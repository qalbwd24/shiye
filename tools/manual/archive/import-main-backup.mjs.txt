// Historical real-data acceptance script. Read tools/manual/README.md first.
// Explicit local migration of the user-provided archive. Never sends data off this machine.
import {createReadStream,promises as fs} from 'node:fs';
const base='http://127.0.0.1:18765',file='F:/我的奋斗/备份下载/20260914/diary_backup_all_20260914_134340_v3.zip';
const before=await(await fetch(base+'/api/library')).json();
if(!Array.isArray(before.entries))throw Error('Library is unavailable or locked');
const request=(p,body,headers={})=>fetch(base+p,{method:'POST',headers:{'X-Diary-Client':'desktop',...headers},body,duplex:'half'});
const start=Date.now(),r=await request('/api/import-upload',createReadStream(file),{'X-File-Name':encodeURIComponent(file.split('/').pop())});
const job=await r.json();if(!r.ok)throw Error(job.error);
console.log(JSON.stringify({stage:'uploaded',bytes:job.bytes}));
const response=await request('/api/import-upload/'+job.id+'/commit','{}',{'Content-Type':'application/json'}),report=await response.json();
if(!response.ok)throw Error(report.error);
const after=await(await fetch(base+'/api/library')).json();
const preserved=before.entries.every(e=>after.entries.some(a=>a.id===e.id));
if(!preserved)throw Error('Preexisting entries missing');
const result={...report,elapsedSeconds:(Date.now()-start)/1000,previousEntries:before.entries.length,totalEntries:after.entries.length,preexistingEntriesPreserved:preserved};
await fs.writeFile('F:/我的奋斗/迁移记录/正式库导入结果.json',JSON.stringify(result,null,2),'utf8');
console.log(JSON.stringify(result));
